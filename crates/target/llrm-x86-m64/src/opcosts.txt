# What flat 32-bit code costs by operation: each MIR operation as a sum of the
# prices of the instruction forms it becomes (`timings.times` has them per CPU).
# Flat code has no far call, no segment to save around a string operation and no
# 66h on a dword; `prefix` is the CPU's price of the one 16-bit forms carry.

# None: the multiply-high is priced at the 486's shortest multiply (`multiply_high`), the dividend being the multiplier.
divide_margin = 0

# SSE converts in one instruction; the long double that uses x87 never reaches a select of conversions.
float_to_int = 0

copy_variable = 0
direction = 2*alu_rr
# The operations a completely unrolled loop may total, before the work its copy no longer does boosts it up to
# fourfold (peelsize). From LLVM's `-unroll-threshold-default` (150; -O3's 300 is twice it), whose 400% boost the
# pass takes. Calibrated on the vsgcc matmul: at 150 its code is 1728 B (clang -O2 1789) and 7.6% slower than at
# GCC's 200 (2633 B), still faster than gcc (17734 clocks) and clang (13144) against 11307; no other program's
# clocks move. Change it by measuring the same two columns again.
unroll_budget = 150
